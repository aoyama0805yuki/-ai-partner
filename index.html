function doPost(e) { let req = {}; try { req = JSON.parse(e.postData.contents); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ error: "無効なリクエストです" })) .setMimeType(ContentService.MimeType.JSON); } const prompt = req.prompt; const history = req.history || []; const sysInst = req.systemInstruction; const GEMINI_API_KEY = PropertiesService.getScriptProperties().getProperty('GEMINI_API_KEY'); if (!GEMINI_API_KEY) { return ContentService.createTextOutput(JSON.stringify({ error: "GAS側にAPIキーが設定されていません。" })) .setMimeType(ContentService.MimeType.JSON); } // ★ 安定・無料枠重視のLiteモデルを使用 ★ const MODEL = 'gemini-3.5-flash-lite'; const url = `https://generativelanguage.googleapis.com/v1beta/models/${MODEL}:generateContent?key=${GEMINI_API_KEY}`; // --- 会話履歴(文脈)をGeminiの形式に変換 --- let contents = []; let lastRole = null; history.forEach((msg, index) => { if (msg.role === 'system') return; let role = msg.role === 'ai' ? 'model' : 'user'; let textToUse = msg.text; if (index === history.length - 1 && role === 'user') { textToUse = prompt; } if (role === lastRole) { contents[contents.length - 1].parts[0].text += "\n\n" + textToUse; } else { contents.push({ role: role, parts: [{ text: textToUse }] }); lastRole = role; } }); if (contents.length === 0) { contents.push({ role: 'user', parts: [{ text: prompt }] }); } const payload = { contents: contents, systemInstruction: { role: 'system', parts: [{ text: sysInst }] }, // tools: [{ googleSearch: {} }], // ★ エラー回避のため検索機能は一旦オフ generationConfig: { responseMimeType: 'application/json' } }; const options = { method: "post", contentType: "application/json", payload: JSON.stringify(payload), muteHttpExceptions: true }; try { const res = UrlFetchApp.fetch(url, options); const data = JSON.parse(res.getContentText()); if (res.getResponseCode() !== 200) { return ContentService.createTextOutput(JSON.stringify({ error: data.error?.message || "APIエラー" })) .setMimeType(ContentService.MimeType.JSON); } const rawText = data?.candidates?.[0]?.content?.parts?.[0]?.text || '{}'; return ContentService.createTextOutput(rawText).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ error: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
